const productGrid = document.querySelector("#product-grid");
const filterButtons = document.querySelectorAll(".filter-button");
const menuToggle = document.querySelector(".menu-toggle");
const primaryNav = document.querySelector("#primary-nav");
const header = document.querySelector(".site-header");

let toastTimer;

/* Render reusable product cards. */
function renderProducts(filter = "all") {
  if (!productGrid) return;

  const visibleProducts = filter === "all"
    ? products
    : products.filter((product) => product.category === filter);

  productGrid.innerHTML = visibleProducts.map((product) => {
    const productIndex = products.indexOf(product);

    return `
      <article class="product-card">
        <div class="product-card-image">
          <img src="${product.image}" alt="${product.alt}" loading="lazy">
          <span class="product-tag">${product.tag}</span>
        </div>
        <div class="product-info">
          <h3>${product.name}</h3>
          <p>${product.description}</p>
          <div class="product-actions">
            <button type="button" data-view-product="${productIndex}">View Details <span aria-hidden="true">↗</span></button>
            <a class="product-enquire" href="#enquiry" data-product-enquiry="${product.name}">Enquire Now <span aria-hidden="true">→</span></a>
          </div>
        </div>
      </article>
    `;
  }).join("");
}

renderProducts();

/* Product category filters. */
filterButtons.forEach((button) => {
  button.addEventListener("click", () => {
    filterButtons.forEach((item) => {
      item.classList.remove("active");
      item.setAttribute("aria-pressed", "false");
    });

    button.classList.add("active");
    button.setAttribute("aria-pressed", "true");
    renderProducts(button.dataset.filter);
  });
});

/* Mobile navigation open/close. */
function closeMenu() {
  if (!menuToggle || !primaryNav) return;
  menuToggle.setAttribute("aria-expanded", "false");
  menuToggle.setAttribute("aria-label", "Open navigation menu");
  primaryNav.classList.remove("is-open");
}

if (menuToggle && primaryNav) {
  menuToggle.addEventListener("click", () => {
    const isExpanded = menuToggle.getAttribute("aria-expanded") === "true";
    menuToggle.setAttribute("aria-expanded", String(!isExpanded));
    menuToggle.setAttribute("aria-label", isExpanded ? "Open navigation menu" : "Close navigation menu");
    primaryNav.classList.toggle("is-open", !isExpanded);
  });

  primaryNav.querySelectorAll("a").forEach((link) => {
    link.addEventListener("click", closeMenu);
  });

  document.addEventListener("click", (event) => {
    if (!primaryNav.contains(event.target) && !menuToggle.contains(event.target)) {
      closeMenu();
    }
  });
}

/* Smooth anchor navigation, with mobile menu close. */
document.querySelectorAll('a[href^="#"]').forEach((link) => {
  link.addEventListener("click", (event) => {
    const targetId = link.getAttribute("href");
    if (!targetId || targetId === "#") return;

    const target = document.querySelector(targetId);
    if (!target) return;

    event.preventDefault();
    closeMenu();
    target.scrollIntoView({ behavior: "smooth", block: "start" });
  });
});

/* Sticky header style and back-to-top visibility. */
function updateScrollUI() {
  const scrolled = window.scrollY > 35;
  header?.classList.toggle("is-scrolled", scrolled);
  backToTop?.classList.toggle("visible", window.scrollY > 550);
}

window.addEventListener("scroll", updateScrollUI, { passive: true });
updateScrollUI();

backToTop?.addEventListener("click", () => {
  window.scrollTo({ top: 0, behavior: "smooth" });
});

/* Scroll reveal animations. */
const revealItems = document.querySelectorAll(".reveal");
if ("IntersectionObserver" in window) {
  const revealObserver = new IntersectionObserver((entries, observer) => {
    entries.forEach((entry) => {
      if (entry.isIntersecting) {
        entry.target.classList.add("is-visible");
        observer.unobserve(entry.target);
      }
    });
  }, { threshold: 0.12 });

  revealItems.forEach((item) => revealObserver.observe(item));
} else {
  revealItems.forEach((item) => item.classList.add("is-visible"));
}

/* Small status notifications. */
function showToast(message) {
  if (!toast) return;
  toast.textContent = message;
  toast.classList.add("show");
  window.clearTimeout(toastTimer);
  toastTimer = window.setTimeout(() => toast.classList.remove("show"), 3200);
}

/* Modal helpers. */
function openModal(modal) {
  if (!modal) return;
  modal.classList.add("is-open");
  modal.setAttribute("aria-hidden", "false");
  document.body.classList.add("modal-open");
}

function closeModal(modal) {
  if (!modal) return;
  modal.classList.remove("is-open");
  modal.setAttribute("aria-hidden", "true");

  if (!document.querySelector(".modal.is-open, .lightbox.is-open")) {
    document.body.classList.remove("modal-open");
  }
}

/* Product details modal, plus enquire actions. */
function openProductDetails(product) {
  const image = document.querySelector("#modal-product-image");
  const title = document.querySelector("#modal-product-title");
  const description = document.querySelector("#modal-product-description");
  const enquireLink = document.querySelector("#modal-enquire-link");

  if (!image || !title || !description || !enquireLink) return;

  image.src = product.image;
  image.alt = product.alt;
  title.textContent = product.name;
  description.textContent = product.description;
  enquireLink.dataset.productEnquiry = product.name;

  openModal(productModal);
}

function selectEnquiryProduct(productName) {
  if (!productSelect || !productName) return;

  const supportedOptions = Array.from(productSelect.options).map((option) => option.value || option.textContent);
  const matchingOption = supportedOptions.find((option) => option.toLowerCase() === productName.toLowerCase());

  if (matchingOption) {
    productSelect.value = matchingOption;
  } else if (/lily/i.test(productName)) {
    productSelect.value = "Lilies";
  } else if (/rose/i.test(productName)) {
    productSelect.value = "Roses";
  } else {
    productSelect.value = "Other";
  }

  closeModal(productModal);
  const enquirySection = document.querySelector("#enquiry");
  enquirySection?.scrollIntoView({ behavior: "smooth", block: "start" });
  window.setTimeout(() => productSelect.focus({ preventScroll: true }), 450);
}

document.addEventListener("click", (event) => {
  const detailsButton = event.target.closest("[data-view-product]");
  if (detailsButton) {
    const product = products[Number(detailsButton.dataset.viewProduct)];
    if (product) openProductDetails(product);
    return;
  }

  const enquiryLink = event.target.closest("[data-product-enquiry], [data-enquiry-product]");
  if (enquiryLink) {
    const productName = enquiryLink.dataset.productEnquiry || enquiryLink.dataset.enquiryProduct;
    if (productName) {
      event.preventDefault();
      selectEnquiryProduct(productName);
    }
  }

  if (event.target.closest("[data-close-modal]")) {
    closeModal(productModal);
  }
});

/* Gallery lightbox. */
document.querySelectorAll(".gallery-item").forEach((item) => {
  item.addEventListener("click", () => {
    const image = document.querySelector("#lightbox-image");
    const caption = document.querySelector("#lightbox-caption");

    if (!image || !caption) return;
    image.src = item.dataset.image || "";
    image.alt = item.querySelector("img")?.alt || "";
    caption.textContent = item.dataset.caption || "";
    openModal(galleryLightbox);
  });
});

function closeLightbox() {
  closeModal(galleryLightbox);
}

galleryLightbox?.querySelector(".lightbox-close")?.addEventListener("click", closeLightbox);
galleryLightbox?.querySelector(".lightbox-backdrop")?.addEventListener("click", closeLightbox);

/* Close open overlays with Escape. */
document.addEventListener("keydown", (event) => {
  if (event.key !== "Escape") return;
  closeModal(productModal);
  closeLightbox();
  closeMenu();
});

/* Demo front-end form validation and success feedback.
   Replace this handler with a fetch() request when a backend is connected. */
enquiryForm?.addEventListener("submit", (event) => {
  event.preventDefault();

  if (!enquiryForm.checkValidity()) {
    enquiryForm.reportValidity();
    return;
  }

  const phoneInput = enquiryForm.elements.phone;
  const digits = phoneInput.value.replace(/\D/g, "");

  if (digits.length < 7) {
    phoneInput.setCustomValidity("Please enter a valid phone number.");
    phoneInput.reportValidity();
    phoneInput.setCustomValidity("");
    return;
  }

  const formData = new FormData(enquiryForm);
  const enquiry = Object.fromEntries(formData.entries());

  // Demo only: inspect the prepared enquiry in the browser console.
  console.info("Demo enquiry ready for backend integration:", enquiry);

  formSuccess.hidden = false;
  enquiryForm.reset();
  showToast("Thank you — your enquiry has been prepared.");
  formSuccess.scrollIntoView({ behavior: "smooth", block: "nearest" });
});

/* WhatsApp floating button and contact link. */
function buildWhatsAppUrl(message) {
  const phone = BUSINESS_CONFIG.whatsappNumber.replace(/\D/g, "");
  return `https://wa.me/${phone}?text=${encodeURIComponent(message)}`;
}

const whatsappButton = document.querySelector("#whatsapp-button");
if (whatsappButton) {
  whatsappButton.href = buildWhatsAppUrl(BUSINESS_CONFIG.whatsappMessage);
}

const contactWhatsApp = document.querySelector(".whatsapp-contact");
if (contactWhatsApp) {
  contactWhatsApp.href = buildWhatsAppUrl(BUSINESS_CONFIG.whatsappMessage);
}
